# Regression: C joins the namespace into a type name with `__`, which a DSDL name may hold as well,
# and declares every type in one global scope. `ns.A__B` and `ns.A.B` were both `ns__A__B`, and
# `ns.Svc`'s request section and `ns.Svc.Request` were both `ns__Svc__Request`, so a translation unit
# including both headers did not compile. The object backend names its C the same way.

# RUN: rm -rf %t && mkdir -p %t/join/ns/A %t/skew/ns/A %t/svc/ns/Svc
# RUN: printf 'uint8 a\n@sealed\n' > %t/join/ns/A__B.1.0.dsdl
# RUN: printf 'uint8 b\n@sealed\n' > %t/join/ns/A/B.1.0.dsdl
# RUN: printf 'uint8 a\n@sealed\n' > %t/skew/ns/A__B.1.0.dsdl
# RUN: printf 'uint8 b\n@sealed\n' > %t/skew/ns/A/B.2.0.dsdl
# RUN: printf 'uint8 a\n@sealed\n---\nuint8 b\n@sealed\n' > %t/svc/ns/Svc.1.0.dsdl
# RUN: printf 'uint8 c\n@sealed\n' > %t/svc/ns/Svc/Request.1.0.dsdl
