# C escapes only what C reserves: `class`, `operator` and `export` are ordinary identifiers here.
# C_KEYWORDS: uint8_t break_;
# C_KEYWORDS: uint8_t func;
# C_KEYWORDS: uint8_t class;
# C_KEYWORDS: uint8_t typedef_;
# C_KEYWORDS: uint8_t operator;
# C_KEYWORDS: uint8_t export;

# Field names reach C verbatim, so the three case variants stay distinct without disambiguation.
# C_FOLDS: uint8_t fooBar;
# C_FOLDS: uint8_t foo_bar;
# C_FOLDS: uint8_t FooBar;

# C++ reserves more than C: `class`, `operator` and `export` are escaped, `impl` and `func` are not.
# CPP_KEYWORDS: std::uint8_t break_{};
# CPP_KEYWORDS: std::uint8_t func{};
# CPP_KEYWORDS: std::uint8_t class_{};
# CPP_KEYWORDS: std::uint8_t typedef_{};
# CPP_KEYWORDS: std::uint8_t operator_{};
# CPP_KEYWORDS: std::uint8_t export_{};

# Rust escapes its own set: `impl`, `let` and `yield` are keywords, `typedef` and `class` are not.
# RUST_KEYWORDS: pub break_: u8,
# RUST_KEYWORDS: pub impl_: u8,
# RUST_KEYWORDS: pub class: u8,
# RUST_KEYWORDS: pub let_: u8,
# RUST_KEYWORDS: pub typedef: u8,
# RUST_KEYWORDS: pub yield_: u8,

# Go exports fields as PascalCase, which lifts every one of them clear of the keyword set: Go
# keywords are lowercase, so no field here needs an escape at all.
# GO_KEYWORDS: type Keywords_1_0 struct {
# GO_KEYWORDS: Break uint8
# GO_KEYWORDS: Func uint8
# GO_KEYWORDS: Map uint8
# GO_KEYWORDS: Range uint8

# The PascalCase projection is many-to-one, so the per-scope allocator disambiguates.
# GO_FOLDS: FooBar uint8
# GO_FOLDS: FooBar_2 uint8
# GO_FOLDS: FooBar_3 uint8

# TypeScript and Python fold fields to snake_case, which puts them back in reach of the keyword set.
# TS_KEYWORDS: break_: number;
# TS_KEYWORDS: class_: number;
# TS_KEYWORDS: interface_: number;
# TS_KEYWORDS: let_: number;
# TS_KEYWORDS: export_: number;

# TS_FOLDS: foo_bar: number;
# TS_FOLDS: foo_bar_2: number;
# TS_FOLDS: foo_bar_3: number;

# The snake_case projection drops a leading underscore but keeps a trailing one, so `_foo` and
# `foo_` survive as distinct names -- by luck, not by design: had the stem been a keyword they
# would have collided with each other's escaped form.
# TS_UNDERSCORES: foo: number;
# TS_UNDERSCORES: foo_: number;
# TS_UNDERSCORES: bar: number;

# PY_KEYWORDS: break_: int = 0
# PY_KEYWORDS: class_: int = 0
# PY_KEYWORDS: def_: int = 0
# PY_KEYWORDS: lambda_: int = 0
# PY_KEYWORDS: yield_: int = 0

# PY_FOLDS: foo_bar: int = 0
# PY_FOLDS: foo_bar_2: int = 0
# PY_FOLDS: foo_bar_3: int = 0

# C builds macro tokens by upper-casing the raw name, so `fooBar` becomes FOOBAR rather than
# FOO_BAR and does not collide with the constant actually spelled FOO_BAR. C also skips the keyword
# escape for macros -- `break` is not a macro-namespace hazard in C -- so this reads BREAK, not
# BREAK_. Both are the `strop: no` cell of the role table in test/unit/golden/naming-roles.txt.
# C_CONSTANTS: #define fixtures_naming__naming__Constants_FOO_BAR (1)
# C_CONSTANTS: #define fixtures_naming__naming__Constants_FOOBAR (2)
# C_CONSTANTS: #define fixtures_naming__naming__Constants_BREAK (3)

# Go folds to UPPER_SNAKE_CASE first, so `FOO_BAR` and `fooBar` do collide and the scope repairs the
# second one; and it does strop, so `break` becomes BREAK_.
# GO_CONSTANTS: const CONSTANTS_1_0_FOO_BAR = 1
# GO_CONSTANTS: const CONSTANTS_1_0_FOO_BAR_2 = 2
# GO_CONSTANTS: const CONSTANTS_1_0_BREAK_ = 3

# GO_NAMESPACE: package map_

# PY_CONSTANTS: CONSTANTS_1_0_FOO_BAR = 1
# PY_CONSTANTS: CONSTANTS_1_0_FOO_BAR_2 = 2
# PY_CONSTANTS: CONSTANTS_1_0_BREAK_ = 3
