# Regression: a definition's output file and a namespace's directory take one module name. In Rust
# `ns/file_1_0.rs` and `ns/file_1_0/mod.rs` are both `mod file_1_0`, which rustc refuses (E0761,
# E0428), and in Python the package `ns/file_1_0/` hides the module `ns/file_1_0.py`. C and C++
# headers keep their extension, Go imports a directory as a package, and TypeScript resolves
# `./file_1_0` to the file and a type in the directory by its own path, so those four generate.

# RUN: rm -rf %t && mkdir -p %t/dsdl/ns/file_1_0
# RUN: printf 'uint8 a\n@sealed\n' > %t/dsdl/ns/File.1.0.dsdl
# RUN: printf 'uint8 b\n@sealed\n' > %t/dsdl/ns/file_1_0/X.1.0.dsdl

# `ast` emits nothing, so it checks every backend and names the ones affected.
# RUN: not %dsdlc --target-language ast %t/dsdl/ns 2>&1 | FileCheck %s --check-prefix=ALL
# ALL: error: type name collision in generated output: ns.File.1.0 and namespace ns.file_1_0, which holds ns.file_1_0.X, map to the same module name for target languages 'rust', 'python'
# ALL-NOT: type name collision

# RUN: not %dsdlc --target-language rust %t/dsdl/ns --outdir %t/rust 2>&1 | FileCheck %s --check-prefix=RUST
# RUST: map to the same module name for target language 'rust'
# RUN: not %dsdlc --target-language python %t/dsdl/ns --outdir %t/py 2>&1 | FileCheck %s --check-prefix=PY
# PY: map to the same module name for target language 'python'

# RUN: %dsdlc --target-language c %t/dsdl/ns --outdir %t/c
# RUN: %dsdlc --target-language cpp --cpp-profile std %t/dsdl/ns --outdir %t/cpp
# RUN: %dsdlc --target-language go --go-module m %t/dsdl/ns --outdir %t/go
# RUN: %dsdlc --target-language ts %t/dsdl/ns --outdir %t/ts
# RUN: ls %t/ts/ns | FileCheck %s --check-prefix=TS
# TS-DAG: file_1_0.ts
# TS-DAG: file_1_0{{$}}

# The namespace need not hold the definition directly: a deeper one reaches the directory as well.
# RUN: rm -rf %t/deep && mkdir -p %t/deep/ns/file_1_0/inner
# RUN: printf 'uint8 a\n@sealed\n' > %t/deep/ns/File.1.0.dsdl
# RUN: printf 'uint8 b\n@sealed\n' > %t/deep/ns/file_1_0/inner/Y.1.0.dsdl
# RUN: not %dsdlc --target-language rust %t/deep/ns --outdir %t/deepout 2>&1 | FileCheck %s --check-prefix=DEEP
# DEEP: ns.File.1.0 and namespace ns.file_1_0, which holds ns.file_1_0.inner.Y, map to the same module name

# Another version of the definition has another file name, and generates.
# RUN: rm -rf %t/other && mkdir -p %t/other/ns/file_1_0
# RUN: printf 'uint8 a\n@sealed\n' > %t/other/ns/File.2.0.dsdl
# RUN: printf 'uint8 b\n@sealed\n' > %t/other/ns/file_1_0/X.1.0.dsdl
# RUN: %dsdlc --target-language rust %t/other/ns --outdir %t/otherout
# RUN: cat %t/otherout/src/ns/mod.rs | FileCheck %s --check-prefix=OTHER
# OTHER-DAG: pub mod file_1_0;
# OTHER-DAG: pub mod file_2_0;
