# Regression: a C++ namespace is declared beside the structs of its parent namespace, so a
# definition and a namespace of one name there are `namespace ns { struct Foo; namespace Foo {} }`,
# which clang refuses ("redefinition of 'Foo' as different kind of symbol"). The type name decides
# which namespace it meets: `Foo` by default, and `Foo_1_0` under --versioned-type-names.

# RUN: rm -rf %t && mkdir -p %t/plain/ns/Foo %t/versioned/ns/Foo_1_0 %t/svc/ns/Svc_Request
# RUN: printf 'uint8 a\n@sealed\n' > %t/plain/ns/Foo.1.0.dsdl
# RUN: printf 'uint8 b\n@sealed\n' > %t/plain/ns/Foo/X.1.0.dsdl
# RUN: printf 'uint8 a\n@sealed\n' > %t/versioned/ns/Foo.1.0.dsdl
# RUN: printf 'uint8 b\n@sealed\n' > %t/versioned/ns/Foo_1_0/X.1.0.dsdl
# RUN: printf 'uint8 a\n@sealed\n---\nuint8 b\n@sealed\n' > %t/svc/ns/Svc.1.0.dsdl
# RUN: printf 'uint8 c\n@sealed\n' > %t/svc/ns/Svc_Request/X.1.0.dsdl

# `ast` emits nothing, so it checks every backend; C++ is the one that declares both in one scope.
