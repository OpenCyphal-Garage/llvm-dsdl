# A vocabulary file binds a concept the generated code needs a library type for to the type a
# profile uses. The C++ span is the first: the built-in binding is std::span for the std and pmr
# profiles, a file replaces it for the profiles it names, and a profile with no binding fails the
# run naming the concept and the flag.

# A binding whose library spells every operation differently from std::span: nothing in the
# emitter assumes the standard shape, and the header includes what the binding names.
# RUN: rm -rf %t && mkdir -p %t
# RUN: %dsdlc --target-language cpp --cpp-profile std --vocabulary %S/vocabulary/acme-byte-view.yaml %S/fixtures_host_image --outdir %t/acme > /dev/null
# RUN: cat %t/acme/fixtures_host_image/vendor/Padded_1_0.hpp | FileCheck %s --check-prefix=ACME
# RUN: %dsdlc --target-language cpp --cpp-profile std --vocabulary %S/vocabulary/acme-byte-view.yaml %S/fixtures_aliasable --outdir %t/acme-nested > /dev/null
# RUN: cat %t/acme-nested/fixtures_aliasable/vendor/Pose_1_0.hpp | FileCheck %s --check-prefix=ACME_NESTED
# ACME:     #include <cstdint>
# ACME-NOT: #include <span>
# ACME:     #include "acme/byte_view.hpp"
# ACME:     static std::uint16_t get_b(const acme::ByteView<const std::uint8_t> buffer)
# ACME-NEXT: {
# ACME-NEXT: const std::size_t buffer_size_bytes = buffer.length();
# ACME-NEXT: dsdl_runtime_get_u16(::llvmdsdl::cpp::readable_bytes(buffer.ptr()),
# ACME:     static std::int8_t set_b(const acme::ByteView<std::uint8_t> buffer, const std::uint16_t member_value)
# ACME:     dsdl_runtime_set_uxx(buffer.ptr(),
# ACME_NESTED: static acme::ByteView<const std::uint8_t> get_orientation(const acme::ByteView<const std::uint8_t> buffer)
# ACME_NESTED: return buffer.tail(

# A member held as a view is the bound span, read and made as the binding spells it.
# RUN: %dsdlc --target-language cpp --cpp-profile std --vocabulary %S/vocabulary/acme-byte-view.yaml --aliasable-views %S/fixtures_views -I %S/fixtures_aliasable --outdir %t/acme-views > /dev/null
# RUN: cat %t/acme-views/fixtures_views/vendor/Frame_1_0.hpp | FileCheck %s --check-prefix=ACME_VIEW
# ACME_VIEW-NOT: #include <span>
# ACME_VIEW:     #include "acme/byte_view.hpp"
# ACME_VIEW:     acme::ByteView<const std::uint8_t> pose;
# ACME_VIEW:     const std::uint8_t* const pose_buf = obj->pose.ptr();
# ACME_VIEW-NEXT: const std::uint64_t pose_size = static_cast<std::uint64_t>(obj->pose.length());
# ACME_VIEW:     out_obj->pose = acme::ByteView<const std::uint8_t>::of(&buf[{{.*}}], static_cast<std::size_t>({{.*}}));

# The file names the std profile alone, so pmr keeps the built-in binding.
# RUN: %dsdlc --target-language cpp --cpp-profile pmr --vocabulary %S/vocabulary/acme-byte-view.yaml %S/fixtures_host_image --outdir %t/pmr > /dev/null
# RUN: cat %t/pmr/fixtures_host_image/vendor/Padded_1_0.hpp | FileCheck %s --check-prefix=PMR
# PMR:     #include <span>
# PMR-NOT: acme
# PMR:     static std::uint16_t get_b(const std::span<const std::uint8_t> buffer)

# The autosar profile has no built-in binding.
# RUN: not %dsdlc --target-language cpp --cpp-profile autosar %S/fixtures_host_image --outdir %t/unbound 2>&1 | FileCheck %s --check-prefix=UNBOUND
# RUN: not test -d %t/unbound/fixtures_host_image
# UNBOUND: the cpp autosar profile has no binding for span: pass --vocabulary <file> with one

# The example binding CETL's span, which has std::span's shape and so names only its type and header.
# RUN: %dsdlc --target-language cpp --cpp-profile autosar --vocabulary %S/../../examples/vocabulary/cpp-autosar-cetl.yaml %S/fixtures_host_image --outdir %t/cetl > /dev/null
# RUN: cat %t/cetl/fixtures_host_image/vendor/Padded_1_0.hpp | FileCheck %s --check-prefix=CETL

# A file is an input of the run.

# INPUTS: acme-byte-view.yaml

# A file that fails validation is reported where it fails, before anything is generated.
