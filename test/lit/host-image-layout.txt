# A host-image type's generated structure is asserted to be the byte image its serialisation
# assumes, on the target the generated code is compiled for: its size, and each member's offset.
# The verdict was decided under natural alignment, and the assertions are where the target's
# compiler confirms it. A type that is not a host image gets none.
#
# Go carries the fold's endianness refusal as a build constraint in a file beside the type's, and
# its folded bodies view the object as bytes through `unsafe`.

# The guard refuses a host it cannot place as well as one it places as big-endian: a compiler
# stating no byte order says nothing about the target. MSVC states none and every target it
# compiles for orders bytes as the wire does, so it is named rather than refused.
# C_GUARD:      #if defined(__BYTE_ORDER__) && defined(__ORDER_LITTLE_ENDIAN__)
# C_GUARD-NEXT: #  if __BYTE_ORDER__ != __ORDER_LITTLE_ENDIAN__
# C_GUARD-NEXT: #    error
# C_GUARD-NEXT: #  endif
# C_GUARD-NEXT: #elif !(defined(_MSC_VER) && (defined(_M_IX86)
# C_GUARD-NEXT: #  error

# The fold is gated on the target triple being little-endian: for a big-endian one the bodies stay
# field-wise, there is no guard to emit, and the layout assertions still hold.
# C_BIG_ENDIAN_BODY-NOT:   dsdl_runtime_image_read(
# C_BIG_ENDIAN_HEADER-NOT: __ORDER_LITTLE_ENDIAN__
# C_BIG_ENDIAN_HEADER:     DSDL_RUNTIME_STATIC_ASSERT(sizeof(struct fixtures_host_image__vendor__Nested) == 12U,

# A host image allocates nothing, so under the PMR profile it carries no memory resource: the
# pointer would make the structure wider than the image, and inside a nested image it would move
# every field after it.
