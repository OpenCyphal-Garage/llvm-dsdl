# Two referenced types that share a short name, and a type that shares its dependency's.
#
# TypeScript and Python bring a referenced type into a file by name, so a definition that reaches
# `collide.alpha.Vec.1.0` and `collide.beta.Vec.1.0` binds `Vec` twice. tsc rejects the duplicate
# outright; Python rebinds it without a word, and both fields are built as the second. The standard
# namespace carries three `Vector3` types and four `Scalar` types in sibling `uavcan.si.unit`
# namespaces, so a definition describing a pose or a battery reaches this.
#
# The first keeps its name and the second takes its namespace, as in Rust.

# RUN: rm -rf %t && mkdir -p %t
# RUN: %dsdlc --target-language ts %S/fixtures_ts_alias/collide --ts-module collide_demo --outdir %t/ts >/dev/null 2>&1
# RUN: cat %t/ts/collide/uses_1_0.ts | FileCheck %s --check-prefix=TS

# The type and both bodies take one local name, so they cannot disagree, and the body and the
# factory refer to it.
# TS: import { deserializeVecFrom, makeVec, serializeVecInto } from "./alpha/vec_1_0";
# TS: import { deserializeVecFrom as deserializeBetaVecFrom, makeVec as makeBetaVec, serializeVecInto as serializeBetaVecInto } from "./beta/vec_1_0";
# TS: import type { Vec } from "./alpha/vec_1_0";
# TS: import type { Vec as BetaVec } from "./beta/vec_1_0";
# TS: from_alpha: Vec;
# TS: from_beta: BetaVec;
# TS: from_alpha: makeVec(),
# TS: from_beta: makeBetaVec(),

# RUN: %dsdlc --target-language python %S/fixtures_ts_alias/collide --outdir %t/py >/dev/null 2>&1
# RUN: cat %t/py/dsdl_gen/collide/uses_1_0.py | FileCheck %s --check-prefix=PY
# PY:      from dsdl_gen.collide.alpha.vec_1_0 import Vec
# PY-NEXT: from dsdl_gen.collide.beta.vec_1_0 import Vec as BetaVec
# PY:      from_alpha: Vec = field(default_factory=lambda: Vec())
# PY:      from_beta: BetaVec = field(default_factory=lambda: BetaVec())
