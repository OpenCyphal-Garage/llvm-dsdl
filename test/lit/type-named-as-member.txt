# A C++ class's members may not take the class's name, and the generated code declares members of
# fixed names in every type. A type named as one of them is declared apart, as `<name>_`, and
# published under the alias `<name>`, as a deprecated type is: a message `ns.FULL_NAME.1.0` is
# `struct FULL_NAME_` and `using FULL_NAME = FULL_NAME_;`. Generated code names the struct, so a
# holder's field type is not shadowed by the holder's own `FULL_NAME`. Rust keeps the name.

# RUN: rm -rf %t && mkdir -p %t/ns
# RUN: printf 'uint8 value\n@sealed\n' > %t/ns/FULL_NAME.1.0.dsdl
# RUN: printf 'uint8 serialize\nuint8 value\n@sealed\n' > %t/ns/serialize.1.0.dsdl
# RUN: printf 'uint8 a\n@sealed\n---\nuint8 b\n@sealed\n' > %t/ns/FIXED_PORT_ID.1.0.dsdl
# RUN: printf '@deprecated\nuint8 value\n@sealed\n' > %t/ns/deserialize.1.0.dsdl
# RUN: printf 'uint8[<=2] value\n@sealed\n' > %t/ns/set_memory_resource.1.0.dsdl
# RUN: printf 'ns.FULL_NAME.1.0 a\nns.serialize.1.0 b\nns.set_memory_resource.1.0 c\n@sealed\n' > %t/ns/Holder.1.0.dsdl
