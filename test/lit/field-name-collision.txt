# Regression: two distinct, spec-valid DSDL field names that differ only in case/underscore style
# (`fooBar` and `foo_bar`) must NOT collide in a generated type. The case-folding backends fold both
# to one identifier -- Go PascalCase `FooBar`, TypeScript/Python snake_case `foo_bar` -- which produced
# a duplicate struct field / object key (Go/TS compile error) or, worst, a silently merged Python
# dataclass attribute so the (de)serialiser read/wrote the wrong field with no error. The fix assigns
# collision-free identifiers per section (a `_2` suffix on the second colliding name). C/C++/Rust
# preserve the raw DSDL name and were already safe. See the naming/identifier sanitization audit.

# RUN: rm -rf %t && mkdir -p %t/dsdl/cx
# The void keeps the bodies field-wise: a host image folds to one move and names no member.
# RUN: printf 'uint8 barBaz = 1\nuint8 bar_baz = 2\nuint8 fooBar\nvoid8\nuint8 foo_bar\n@sealed\n' > %t/dsdl/cx/C.1.0.dsdl

# Go: two distinct exported fields, each referenced by the matching (de)serialise step.
# RUN: %dsdlc --target-language go %t/dsdl --outdir %t/go --go-module cx
# RUN: cat %t/go/dsdl/cx/c_1_0.go | FileCheck %s --check-prefix=GO
# GO-DAG: FooBar uint8
# GO-DAG: FooBar2 uint8
# GO-DAG: c.FooBar
# GO-DAG: c.FooBar2
# Constants fold the same way (barBaz / bar_baz -> CBarBaz); they must stay distinct too.
# GO-DAG: CBarBaz = 1
# GO-DAG: CBarBaz2 = 2

# TypeScript: two distinct properties.
# RUN: %dsdlc --target-language ts %t/dsdl --outdir %t/ts --ts-module cx
# RUN: cat %t/ts/dsdl/cx/c_1_0.ts | FileCheck %s --check-prefix=TS
# TS-DAG: foo_bar: number;
# TS-DAG: foo_bar_2: number;
# TS-DAG: obj.foo_bar_2 =
# TS-DAG: C_BAR_BAZ = 1;
# TS-DAG: C_BAR_BAZ_2 = 2;

# Python: two distinct dataclass attributes (no silent merge).
# RUN: %dsdlc --target-language python %t/dsdl --outdir %t/py
# RUN: cat %t/py/dsdl_gen/dsdl/cx/c_1_0.py | FileCheck %s --check-prefix=PY
# PY-DAG: foo_bar: int = 0
# PY-DAG: foo_bar_2: int = 0
# PY-DAG: self.foo_bar_2 =
# PY-DAG: C_BAR_BAZ = 1
# PY-DAG: C_BAR_BAZ_2 = 2

# Rust folds field names and constants the way the other module-scoped languages do: `fooBar` and
# `foo_bar` reach one identifier and the scope separates them.
# RUN: %dsdlc --target-language rust %t/dsdl --outdir %t/rust --rust-crate-name cx
# RUN: cat %t/rust/src/dsdl/cx/c_1_0.rs | FileCheck %s --check-prefix=RUST
# RUST-DAG: pub const BAR_BAZ: u64 = 1;
# RUST-DAG: pub const BAR_BAZ_2: u64 = 2;
# RUST-DAG: pub foo_bar: u8,
# RUST-DAG: pub foo_bar_2: u8,

# C preserves the raw DSDL names (already collision-free, distinct members).
# RUN: %dsdlc --target-language c %t/dsdl --outdir %t/c
# RUN: cat %t/c/dsdl/cx/C_1_0.h | FileCheck %s --check-prefix=C
# C-DAG: fooBar;
# C-DAG: foo_bar;
