# RUN: %dsdlc --target-language mlir +uavcan.node.Heartbeat.1.0 2>/dev/null | FileCheck %s --check-prefix=MLIR_PLUS

# Listing: outputs are real files, inputs are not -- a synthetic path must never reach a build
# system's DEPENDS set, so a purely embedded target contributes no inputs at all.
# RUN: %dsdlc --target-language c --list-outputs +uavcan.node.Heartbeat.1.0 2>/dev/null | FileCheck %s --check-prefix=PLUS_OUTPUTS
# RUN: %dsdlc --target-language c --list-inputs +uavcan.node.Heartbeat.1.0 >%t.inputs 2>/dev/null
# RUN: not test -s %t.inputs

# The dependency that does get recorded is the binary carrying the catalog.

# Nothing resolves silently to zero targets. Each of these must be a diagnostic, because a build
# that succeeds while generating nothing is the failure this syntax exists to prevent.
# RUN: not %dsdlc --target-language c --outdir %t.typo +uavcna 2>&1 | FileCheck %s --check-prefix=TYPO_NS

# The sigil is target-token syntax, not option syntax, so `--` does not disable it...
# RUN: %dsdlc --target-language c --list-outputs -- +uavcan.node.Heartbeat.1.0 2>/dev/null | FileCheck %s --check-prefix=PLUS_OUTPUTS
# ...and a file literally named '+x' is reached as './+x', which is a path and must fail as one.
# RUN: not %dsdlc --target-language c --outdir %t.escape ./+uavcan 2>&1 | FileCheck %s --check-prefix=ESCAPED_PATH

# Empty-root colon syntax stays an error. '+' was chosen over spelling the builtin root as ':name'
# precisely so a build system that drops a variable fails here instead of silently resolving to the
# embedded catalog and shipping the wrong definition.
# RUN: not %dsdlc --target-language c --outdir %t.colon :uavcan/node/Heartbeat.1.0.dsdl 2>&1 | FileCheck %s --check-prefix=EMPTY_ROOT
